package org.portage.authorizationserver.crypto;

import com.nimbusds.jose.jwk.*;
import com.nimbusds.jose.jwk.gen.ECKeyGenerator;
import com.nimbusds.jose.jwk.source.JWKSource;
import com.nimbusds.jose.proc.SecurityContext;
import java.io.IOException;
import java.nio.file.*;
import java.security.*;
import java.util.*;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.beans.factory.annotation.Value;
import org.portage.authorizationserver.error.InvalidIdentityConfigurationException;

/** Loads or creates local P-256 signing keys. JWK serialization on disk is private; JWKS filters it out. */
@Configuration
public class SigningKeysConfiguration {
    @Bean JWKSource<SecurityContext> jwkSource(@Value("${platform.keys.directory:runtime-keys}") String directory) throws Exception {
        Path activePath = Path.of(directory, "active.jwk");
        Path nextPath = Path.of(directory, "next.jwk");
        ECKey active = loadOrCreate(activePath, "active");
        ECKey next = loadOrCreate(nextPath, "next");
        return (selector, context) -> selector.select(new JWKSet(List.of(active, next)));
    }
    private static ECKey loadOrCreate(Path path, String kid) throws Exception {
        if (Files.exists(path)) {
            try {
                ECKey key = ECKey.parse(Files.readString(path));
                if (!Curve.P_256.equals(key.getCurve()) || !key.isPrivate() || !kid.equals(key.getKeyID()))
                    throw new InvalidIdentityConfigurationException("Invalid signing key curve, identifier or missing private key");
                return key;
            }
            catch (java.text.ParseException | IOException e) { throw new InvalidIdentityConfigurationException("Invalid local signing key file"); }
        }
        var pair = KeyPairGenerator.getInstance("EC");
        pair.initialize(new java.security.spec.ECGenParameterSpec("secp256r1"));
        var keys = pair.generateKeyPair();
        var jwk = new ECKey.Builder(Curve.P_256, (java.security.interfaces.ECPublicKey) keys.getPublic())
                .privateKey((java.security.interfaces.ECPrivateKey) keys.getPrivate()).keyID(kid).algorithm(com.nimbusds.jose.JWSAlgorithm.ES256).build();
        Files.createDirectories(path.toAbsolutePath().getParent());
        Path temporary = Files.createTempFile(path.getParent(), "key-", ".tmp");
        try {
            Files.writeString(temporary, jwk.toJSONString(), StandardOpenOption.TRUNCATE_EXISTING);
            try { Files.move(temporary, path, StandardCopyOption.ATOMIC_MOVE); }
            catch (AtomicMoveNotSupportedException e) { Files.move(temporary, path); }
        } finally { Files.deleteIfExists(temporary); }
        return jwk;
    }
}
